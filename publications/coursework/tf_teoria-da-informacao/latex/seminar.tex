\begin{frame}{Problema \& objetivo}
  \textbf{O trabalho anterior mede a contaminação ambiente$\rightarrow$slack com um $R^2$ linear $= \valRsqLinear$.}
  \begin{itemize}
    \item Captura \alert{apenas} dependência linear --- mas a física é não-linear
          (Arrhenius em $\Tdie$, super-linear em $\Vcore$).
    \item ``Variância explicada'' \alert{não tem unidade transferível}.
  \end{itemize}
  \vfill
  \begin{block}{Objetivo}
    Tratar o sensor de \emph{slack} como um \textbf{canal de comunicação ruidoso}
    e reexpressar ruído, correção e recuperabilidade do envelhecimento \textbf{em bits}.
  \end{block}
\end{frame}

\begin{frame}{\textbf{$\bigstar$ Informação mútua substitui $R^2$}}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}% [TFinfo] era 0.52 — herói: mais espaço p/ figura
      \footnotesize
      \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}\setlength{\topsep}{0pt}
      \begin{itemize}
        \item $\MI{\slk}{\Tdie,\Vcore} = \alert{\valMIstv}$~bits (\emph{binning},
              série bruta, pareada com o $R^2$ bruto), $\valKSGstv$~bits (KSG)
              --- ICs se sobrepõem só estreitamente: uma diferença
              sistemática caracterizada, não concordância.

      envelhecimento/ambiente, não uma não-linearidade instantânea
      demonstrada. Foi o próprio protocolo que encontrou e corrigiu a

\begin{frame}{O pilar da honestidade: $\neff$}
  \begin{itemize}
    \item \valPostSoakSamples{} amostras brutas é \alert{enganoso}:
          $\tauint = \valTauInt$~s ($\sim$5\,min de decorrelação).
    \item Tamanho efetivo de amostra $\;\neff = \alert{\valNeff}\;$ (não 768{,}884).
    \item Todo intervalo de confiança, o piso \emph{surrogate} e o ajuste
          Bayesiano usam $\neff$ --- nunca a contagem bruta.
  \end{itemize}
  \vfill
  \begin{alertblock}{Por que importa}
    A série 1\,Hz é fortemente autocorrelacionada. Reportar sobre $\neff$ com
    ICs por \emph{block-bootstrap} + um nulo \emph{surrogate} é a espinha
    dorsal metodológica. \hfill [\textbf{N4} / fundamentos]
  \end{alertblock}
\end{frame}

\guidequestion{quando o envelhecimento aparece?}% [TFinfo] fio condutor, slides 8–8b
\begin{frame}{Limite de detectabilidade de RUL}
  \begin{itemize}
    \item \textbf{Cram\'er--Rao:} taxa mín.\ detectável \alert{\valMinDetectRate~m-cnt/h};
          tempo mín.\ de observação \alert{\valMinObsTime~h}
          (execuções anteriores de $\sim$23\,h não separavam o envelhecimento).
    \item \textbf{Bayesiano:} inclinação $\valBayesSlope$~m-cnt/h,
          IC 95\% $[\valBayesSlopeLo,\valBayesSlopeHi]$;
          $P(\text{inclinação} < 0) \approx \valBayesPaging\,\%$.
    \item Frequentista $=$ Bayesiano $\Rightarrow$ robusto.
  \end{itemize}
  \vfill
  \begin{block}{Enquadramento}
    Um limite, limitado por ruído, sobre \emph{quando} o envelhecimento se
    torna observável --- \textbf{não} uma predição de vida útil.
    \hfill [\textbf{N4}: CRLB, Bayesiano]
  \end{block}
\end{frame}

\begin{frame}{RUL como distribuição: Wiener/Kalman}
  \footnotesize
  \begin{itemize}

          $\neff$ pontos, suavização Kalman/RTS de uma trajetória de

\begin{frame}{Ressalvas de honestidade}
  \begin{TFhonest}% [TFinfo] painel de honestidade deliberada
  \footnotesize
  \begin{itemize}
    \item O excesso não-linear de MI na série bruta \alert{não sobrevive ao
          \emph{detrend}} --- concentrado na tendência compartilhada
          envelhecimento/ambiente, não em não-linearidade PVT instantânea
          demonstrada.
    \item Um único dispositivo de envelhecimento \& regime único
          $\Rightarrow$ RUL é um \emph{limite} de detectabilidade, não uma predição validada.
    \item $\neff \ll n$ --- toda inferência sobre o tamanho efetivo de amostra.
    \item Capacidade é uma \emph{heurística} de resolução efetiva.
    \item ICA é corroborativa, não uma desmistura limpa.
    \item A comparação KL é um \alert{dispositivo separado} --- nunca misturado com o Dispositivo A.
    \item A distribuição de RUL Wiener/Kalman usa limiares
          auto-referenciados (não existe limiar de falha calibrado para
          este sensor) --- uma metodologia, não uma predição de vida útil.
  \end{itemize}
  \end{TFhonest}
\end{frame}

\begin{frame}[shrink=6]{Contribuições \& conclusão}
  \small
  \begin{itemize}
    \setlength{\itemsep}{1pt}
    \item Uma \textbf{métrica de contaminação em bits} (MI), sem hipótese de
          forma funcional e com unidade transferível entre sensores e bancadas
          --- o excesso bruto acompanha a tendência compartilhada, não uma
          não-linearidade instantânea.
    \item Uma \textbf{definição principiada de qualidade de \emph{burn-in}} $=$
          minimizar $\MI{\slk}{\Tdie,\Vcore}$, junto com um protocolo
          objetivo.
    \item Uma \textbf{especificação informacional do sensor}: $\valCapacityStates$
          estados resolvíveis, $\valBSCcap$~bits por decisão diária, e
          detectabilidade em $\valMinObsTime$~h --- explicando por que
          campanhas de $\sim$23\,h nunca foram suficientes.
    \item Um \textbf{limite honesto de RUL} --- estendido a uma
          \textbf{distribuição de RUL} completa via Wiener/Kalman --- e uma
          \textbf{recuperação de envelhecimento livre de modelo} (ICA).
  \end{itemize}
  \vfill
  \begin{alertblock}{Conclusão}
    $R^2$ diz que o ambiente explica $R^2=\valRsqLinear$ da variância. A
    teoria da informação diz que ele escreve \textbf{\valMIstv~bits} em
    cada leitura --- e, do acoplamento reversível
    ($\valPreMIrev\to\valResidMI$~bits), removemos
    \textbf{\valMIreductionPct\,\%}. Bits são transferíveis; frações de
    variância não são.
  \end{alertblock}
\end{frame}

\begin{frame}{Backup --- ``Cadê a capacidade bang-bang vs.\ PID?''}
  \begin{itemize}
    \item Proposta original: capacidade \emph{por regime térmico}
          (bang-bang vs.\ PID).
    \item Não existe registro bruto de bang-bang sincronizado e comparável
          à campanha PID de \valDurationHours~h $\Rightarrow$ não entregue
          como originalmente proposto.
    \item \textbf{Entregue em seu lugar:} capacidade \& limite de RUL no
          regime PID (este deck); a lacuna distribucional KL/JS entre as
          duas bancadas ($\valKLsp$~bits, JS $=\valJSbench$) como
          sucedâneo direto --- já mostra que a bancada não-PID é mais ruidosa.
    \item \textbf{Primeiro item de trabalho futuro:} adquirir um registro
          bang-bang pareado para completar a comparação de capacidade por regime.
  \end{itemize}
\end{frame}

\begin{frame}{Backup --- déficit de MI detrendizada: como responder}
  \begin{itemize}
    \item ``Como a MI pode ser \emph{menor} que o equivalente do $R^2$?''
    \item Equivalente Gaussiano é um construto de variáveis \emph{contínuas};
          para um resíduo quantizado em poucas contagens, a MI discreta
          pode legitimamente ficar abaixo dele (teto entrópico do resíduo
          + viés descendente do estimador em MI pequena/$\neff$ reduzido).
    \item Leitura defensável: \textbf{nenhum} excesso não-linear é
          resolvível acima do piso do estimador nesta série --- não que o
          acoplamento reversível seja ``sub-linear''.
    \item Dois estimadores independentes (\emph{binning} e KSG) concordam
          no sinal do déficit ($\valMIexcessPctDetrended\,\%$ e
          $\valMIexcessPctDetrKSG\,\%$).
  \end{itemize}
\end{frame}
